let elementos = []; // aquí se guardarán todos los elementos

// Consumir la API
fetch("http://20.104.66.244/api.php")
  .then((response) => response.json())
  .then((data) => {
    elementos = data;
    mostrarTabla(elementos);
  })
  .catch((error) => console.error("Error:", error));

// Función auxiliar: mapear categoría a clase CSS
function clasePorCategoria(categoriaApi) {
  const cat = (categoriaApi || "").toLowerCase();
  if (cat.includes("alcalino")) return "metal-alcalino";
  if (cat.includes("alcalinoterreo")) return "metal-alcalinoterreo";
  if (cat.includes("transicion")) return "metales-de-transicion";
  if (cat.includes("metaloide")) return "metaloide";
  if (cat.includes("no metal")) return "no-metal";
  if (cat.includes("halogeno")) return "halogeno";
  if (cat.includes("gas noble")) return "gas-noble";
  if (cat.includes("lantanido")) return "lantanido";
  if (cat.includes("actinido")) return "actinido";
  return "no-metal"; // por defecto
}
//filtrador de categorias de elementos quimicos
function filtrarCategoria(categoria) {
  const filtrados = elementos.filter(
    (el) => clasePorCategoria(el.categoria) === categoria,
  );
  mostrarTabla(filtrados);
}

// Mostrar la tabla
function mostrarTabla(lista) {
  const contenedor = document.querySelector(".tabla-periodica");
  contenedor.innerHTML = "";

  lista.forEach((el) => {
    const celda = document.createElement("div");
    celda.classList.add("elemento", clasePorCategoria(el.categoria));

    // Posición en la cuadrícula
    if (el.categoria.toLowerCase().includes("lantanido")) {
      celda.style.gridRow = 8;
      celda.style.gridColumn = el.numero_atomico - 54; // La (57) empieza en col 1
    } else if (el.categoria.toLowerCase().includes("actinido")) {
      celda.style.gridRow = 9;
      celda.style.gridColumn = el.numero_atomico - 86; // Ac (89) empieza en col 1
    } else {
      celda.style.gridRow = el.periodo;
      celda.style.gridColumn = el.grupo;
    }

    // Contenido de la celda
    celda.innerHTML = `
      <div class="numero">${el.numero_atomico}</div>
      <div class="simbolo">${el.simbolo}</div>
      <div class="nombre">${el.nombre}</div>
    `;

    // Evento: abrir modal
    celda.addEventListener("click", () => abrirModalDetalle(el));
    contenedor.appendChild(celda);
  });
}

// Buscar elemento por símbolo
function buscarElemento() {
  const simbolo = document
    .getElementById("buscador")
    .value.trim()
    .toLowerCase();
  const resultado = elementos.find(
    (el) => el.simbolo.toLowerCase() === simbolo,
  );

  // Limpiar resaltados previos
  document
    .querySelectorAll(".elemento.resaltado")
    .forEach((n) => n.classList.remove("resaltado"));

  if (resultado) {
    const celda = [...document.querySelectorAll(".elemento")].find(
      (c) => c.querySelector(".simbolo").textContent === resultado.simbolo,
    );

    if (celda) {
      celda.classList.add("resaltado");
      celda.scrollIntoView({ behavior: "smooth", block: "center" });
    }
    abrirModalDetalle(resultado);
  } else {
    alert("No encontrado");
  }
}

// Abrir modal
function abrirModalDetalle(elemento) {
  document.getElementById("detalle-simbolo").textContent = elemento.simbolo;
  document.getElementById("detalle-nombre").textContent =
    `Nombre: ${elemento.nombre}`;
  document.getElementById("detalle-numero").textContent =
    `Número atómico: ${elemento.numero_atomico}`;
  document.getElementById("detalle-categoria").textContent =
    `Categoría: ${elemento.categoria}`;
  document.getElementById("detalle-peso").textContent =
    `Peso atómico: ${elemento.peso_atomico || "N/A"}`;
  document.getElementById("detalle-descubridor").textContent =
    `Descubridor: ${elemento.descubridor || "Desconocido"}`;
  document.getElementById("detalle-config").textContent =
    `Configuración electrónica: ${elemento.configuracion_electronica || "N/A"}`;

  const header = document.getElementById("modal-header");
  header.textContent = elemento.nombre;
  header.className = "modal-header " + clasePorCategoria(elemento.categoria);

  const img = document.getElementById("detalle-imagen");
  img.src =
    elemento.imagen ||
    "https://via.placeholder.com/150?text=" + elemento.simbolo;

  const modal = document.getElementById("modal-detalle");
  modal.classList.remove("modal-hidden");
  modal.style.display = "flex";
}

// Cerrar modal (clic fuera, botón, ESC)
document.addEventListener("click", (e) => {
  const modal = document.getElementById("modal-detalle");
  if (!modal) return;
  const contenido = modal.querySelector(".modal-contenido");
  if (
    e.target.id === "modal-cerrar" ||
    (e.target === modal && !contenido.contains(e.target))
  ) {
    modal.classList.add("modal-hidden");
    modal.style.display = "none";
  }
});

document.addEventListener("keydown", (e) => {
  if (e.key === "Escape") {
    const modal = document.getElementById("modal-detalle");
    modal.classList.add("modal-hidden");
    modal.style.display = "none";
  }
});
